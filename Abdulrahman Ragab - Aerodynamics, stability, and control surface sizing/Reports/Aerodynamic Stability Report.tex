\section{Horizontal Tail Sizing \& Optimum Length Calculation}
% ============================================================

\begin{itemize}
    \item $V_{HT} = 0.7$
    \item ($\text{MAC} = 0.13\,\text{m}$
    \item $S_w = 0.08001\,\text{m}^2$
\end{itemize}
  

\begin{itemize}
    \item \textbf{Fuselage cross section Length ($L_f$):} $0.08\,\text{m}$
    \item \textbf{Equivalent Fuselage Diameter ($D_f$):} Evaluated via cross-sectional area $A_f$ as:
    \begin{equation}
        D_f = \sqrt{\frac{4 A_f}{\pi}} = 0.096\,\text{m}
    \end{equation}
    \item \textbf{Optimum Tail Arm Length ($L_{opt}$):} Derived from stability requirements and body geometry:
    \begin{equation}
        L_{opt} = \sqrt{\frac{4 \cdot \text{MAC} \cdot S_w \cdot V_{HT}}{\pi \cdot D_f}} = 0.311\,\text{m}
    \end{equation}
    \item \textbf{Required Horizontal Tail Area ($S_{HT}$):} Calculated using the tail volume relation:
    \begin{equation}
        S_{HT} = \frac{V_{HT} \cdot \text{MAC} \cdot S_w}{L_{opt}} = 0.0234\,\text{m}^2
    \end{equation}
    \item \textbf{Aspect Ratio ($AR_{ht}$):} Scaled relative to the main wing aspect ratio ($AR_w$):
    \begin{equation}
        AR_{ht} = \frac{2}{3} AR_w = 3.306
    \end{equation}
    \item \textbf{Horizontal Tail Wingspan ($b_{ht}$):} Derived from the aspect ratio and surface area:
    \begin{equation}
        b_{ht} = \sqrt{AR_{ht} \cdot S_{HT}} = 0.278\,\text{m}
    \end{equation}
\end{itemize}

\section{Neutral Point and Static Margin}
% ============================================================

The neutral point calculated by AVL was $X_{NP}=0.103210~\mathrm{m}$. With $X_{CG}=0.07880~\mathrm{m}$ and $c_{\mathrm{ref}}=0.12969~\mathrm{m}$, the static margin is:
\begin{equation}
SM = \frac{X_{NP}-X_{CG}}{c_{\mathrm{ref}}} = \frac{0.103210-0.07880}{0.12969} \approx 0.188 \implies \boxed{SM\approx18.8\%}
\end{equation}

\subsection{Dynamic Mode Identification}
The eigenvalue spectrum corresponds to five principal aircraft dynamic modes:
\begin{enumerate}
    \item \textbf{Roll Subsidence:} $\lambda = -23.800234$ (Stable, time constant $\tau \approx 0.0420\,\text{s}$).
    \item \textbf{Dutch Roll:} $\lambda = -1.5364 \pm 13.0286i$ ($\omega_n = 13.119\,\text{rad/s}$, $\zeta = 0.117$, stable but lightly damped).
    \item \textbf{Short-Period Mode:} $\lambda = -16.7756 \pm 17.9719i$ ($\omega_n = 24.583\,\text{rad/s}$, $\zeta = 0.682$, stable and strongly damped).
    \item \textbf{Spiral Mode:} $\lambda = +0.09608$ (Unstable, doubling time $t_2 \approx 7.22\,\text{s}$).
    \item \textbf{Phugoid:} $\lambda = +0.00339 \pm 1.3367i$ ($\zeta = -0.00254$, very slightly unstable, doubling time $\approx 204.2\,\text{s}$).
\end{enumerate}
